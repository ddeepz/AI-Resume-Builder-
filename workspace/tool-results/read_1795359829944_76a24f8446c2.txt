    26→function PhotoView({
    27→  photo,
    28→  style,
    29→  size = 90,
    30→  template,
    31→}: {
    32→  photo: string;
    33→  style: PhotoStyle;
    34→  size?: number;
    35→  template: TemplateConfig;
    36→}) {
    37→  if (!photo) return null;
    38→  const borderRadius =
    39→    style === "circular" ? "50%" : style === "rounded" ? "12px" : "0px";
    40→  return (
    41→    <img
    42→      src={photo}
    43→      alt="Profile"
    44→      style={{
    45→        width: `${size}px`,
    46→        height: `${size}px`,
    47→        objectFit: "cover",
    48→        borderRadius,
    49→        border: `3px solid ${template.accent}`,
    50→      }}
    51→    />
    52→  );
    53→}
