   203→/* ---------- Step 3: Skills ---------- */
   204→function TagInput({
   205→  type,
   206→  label,
   207→}: {
   208→  type: "technical" | "programming" | "soft";
   209→  label: string;
   210→}) {
   211→  const { data, addSkill, removeSkill } = useResumeStore();
   212→  const [value, setValue] = useState("");
   213→  const inputRef = useRef<HTMLInputElement>(null);
   214→
   215→  const onKey = (e: React.KeyboardEvent<HTMLInputElement>) => {
   216→    if (e.key === "Enter" || e.key === ",") {
   217→      e.preventDefault();
   218→      const v = value.trim().replace(",", "");
   219→      if (v) {
   220→        addSkill(type, v);
   221→        setValue("");
   222→      }
   223→    }
   224→  };
   225→
   226→  return (
   227→    <div className="nx-input-group">
   228→      <Label>{label}</Label>
   229→      <div className="nx-tag-container" onClick={() => inputRef.current?.focus()}>
   230→        {data.skills[type].map((s, idx) => (
   231→          <span className="nx-tag" key={s + idx}>
   232→            {s}
   233→            <button
   234→              className="nx-tag-remove"
   235→              onClick={(e) => {
   236→                e.stopPropagation();
   237→                removeSkill(type, idx);
   238→              }}
   239→              type="button"
   240→              aria-label={`Remove ${s}`}
   241→            >
   242→              ×
   243→            </button>
   244→          </span>
   245→        ))}
   246→        <input
   247→          ref={inputRef}
   248→          className="nx-tag-input"
   249→          placeholder="Type and press Enter..."
   250→          value={value}
   251→          onChange={(e) => setValue(e.target.value)}
   252→          onKeyDown={onKey}
   253→        />
   254→      </div>
   255→    </div>
   256→  );
   257→}

   259→export function SkillsStep() {
   260→  const { data, addSkill } = useResumeStore();
   261→  const toast = useToast();
   262→  const [loading, setLoading] = useState(false);
   263→  const [suggestions, setSuggestions] = useState<{ technical: string[]; soft: string[] } | null>(null);
   264→
   265→  const generate = async () => {
   266→    const jobTitle = data.personal.jobTitle.trim();
   267→    if (!jobTitle) {
   268→      toast("Please enter a job title in the Personal Information step first.", "info");
   269→      return;
   270→    }
   271→    setLoading(true);
   272→    try {
   273→      const res = await fetch("/api/ai/skills", {
   274→        method: "POST",
   275→        headers: { "Content-Type": "application/json" },
   276→        body: JSON.stringify({ jobTitle }),
   277→      });
   278→      const json = await res.json();
   279→      if (!res.ok) throw new Error(json.error || "Failed to generate suggestions");
   280→      setSuggestions(json);
   281→      toast("AI skill suggestions generated!", "success");
   282→    } catch (err) {
   283→      toast(err instanceof Error ? err.message : "Failed to generate suggestions", "error");
   284→    } finally {
   285→      setLoading(false);
   286→    }
   287→  };
   288→
   289→  const addSuggested = (skill: string) => {
   290→    const isSoft = suggestions?.soft.includes(skill);
   291→    const exists =
   292→      data.skills.technical.includes(skill) ||
   293→      data.skills.programming.includes(skill) ||
   294→      data.skills.soft.includes(skill);
   295→    if (exists) {
   296→      toast("Skill already added!", "info");
   297→      return;
   298→    }
   299→    if (isSoft) addSkill("soft", skill);
   300→    else {
   301→      const progSkills = ["JavaScript", "Python", "TypeScript", "Java", "Go", "SQL", "HTML5", "CSS3", "R"];
   302→      if (progSkills.includes(skill)) addSkill("programming", skill);
   303→      else addSkill("technical", skill);
   304→    }
   305→    toast(`Added: ${skill}`, "success");
   306→  };
   307→
   308→  return (
   309→    <div className="nx-form-step active">
   310→      <FormHeader icon={Code} title="Skills" subtitle="Showcase your technical and soft skills" />
   311→      <TagInput type="technical" label="Technical Skills" />
   312→      <TagInput type="programming" label="Programming Languages" />
   313→      <TagInput type="soft" label="Soft Skills" />
   314→
   315→      <div className="nx-ai-panel nx-holo-border" style={{ marginTop: "20px" }}>
   316→        <div className="nx-ai-header">
   317→          <div className="nx-ai-icon">
   318→            <Wand2 size={20} color="var(--nx-navy)" />
   319→          </div>
   320→          <div>
   321→            <h4>AI SKILL SUGGESTIONS</h4>
   322→            <p>Based on your target job title</p>
   323→          </div>
   324→        </div>
   325→        <div id="skillSuggestions">
   326→          {!suggestions && !loading && (
   327→            <p style={{ fontSize: "13px", color: "rgba(248,253,255,0.5)" }}>
   328→              Enter a job title in Personal Info to get AI-powered skill suggestions.
   329→            </p>
   330→          )}
   331→          {loading && (
   332→            <div className="nx-typing-indicator">
   333→              <div className="nx-typing-dot" />
   334→              <div className="nx-typing-dot" />
   335→              <div className="nx-typing-dot" />
   336→            </div>
   337→          )}
   338→          {suggestions && !loading && (
   339→            <>
   340→              <p style={{ fontSize: "12px", color: "var(--nx-ice)", marginBottom: "12px" }}>
   341→                Recommended for <strong>{data.personal.jobTitle}</strong>:
   342→              </p>
   343→              <div style={{ display: "flex", flexWrap: "wrap", gap: "6px" }}>
   344→                {[...suggestions.technical, ...suggestions.soft].map((skill) => (
   345→                  <button
   346→                    key={skill}
   347→                    type="button"
   348→                    onClick={() => addSuggested(skill)}
   349→                    style={{
   350→                      padding: "6px 12px",
   351→                      background: "rgba(34,211,238,0.1)",
   352→                      border: "1px solid rgba(34,211,238,0.3)",
   353→                      borderRadius: "20px",
   354→                      fontSize: "12px",
   355→                      color: "var(--nx-ice)",
   356→                      cursor: "pointer",
   357→                      transition: "all 0.3s",
   358→                      display: "inline-flex",
   359→                      alignItems: "center",
   360→                      gap: "4px",
   361→                    }}
   362→                  >
   363→                    <PlusCircle size={9} /> {skill}
   364→                  </button>
   365→                ))}
   366→              </div>
   367→            </>
   368→          )}
   369→        </div>
   370→        <button className="nx-ai-action-btn" onClick={generate} disabled={loading} type="button">
   371→          <Sparkles size={14} />
   372→          {loading ? "Generating..." : "Generate Suggestions"}
   373→        </button>
   374→      </div>
   375→    </div>
   376→  );
   377→}

   566→/* ---------- Step 7: Review & Export ---------- */
   567→export function ReviewStep() {
   568→  const { data, setSummary, setAts, selectedTemplate, photoStyle } = useResumeStore();
   569→  const toast = useToast();
   570→  const [loading, setLoading] = useState(false);
   571→  const [done, setDone] = useState(false);
   572→  const [exporting, setExporting] = useState(false);
   573→  const [aiLoading, setAiLoading] = useState(false);
   574→
   575→  const generate = async () => {
   576→    setLoading(true);
   577→    setAiLoading(true);
   578→    try {
   579→      const res = await fetch("/api/ai/summary", {
   580→        method: "POST",
   581→        headers: { "Content-Type": "application/json" },
   582→        body: JSON.stringify({
   583→          name: data.personal.name,
   584→          jobTitle: data.personal.jobTitle,
   585→          skills: [...data.skills.technical, ...data.skills.programming],
   586→          experience: data.experience,
   587→        }),
   588→      });
   589→      const json = await res.json();
   590→      if (!res.ok) throw new Error(json.error || "Failed to generate summary");
   591→      setSummary(json.summary);
   592→      setDone(true);
